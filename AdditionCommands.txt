/*
The column masks we applied earlier conflict with SINGLE_USER cluster mode. Tables with column masks require USER_ISOLATION mode, which needs multi-node clusters (more expensive).
Also important: your silver transformation uses mode("overwrite") with overwriteSchema, which would drop the masks on each run anyway.
Best approach: remove the masks from the table, and re-apply them as the last step in the pipeline after all writes are done.
*/

-- Drop masks from silver.contributors
ALTER TABLE pfin_dev.silver.contributors ALTER COLUMN contributor_name DROP MASK;
